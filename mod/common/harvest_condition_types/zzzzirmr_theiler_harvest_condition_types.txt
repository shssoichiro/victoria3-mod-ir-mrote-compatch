# Use IR orchards for codling moth; drop only the unreachable potato-PM bonus.

REPLACE:theiler_codling_moth_condition = {
	trigger = {
		theiler_state_can_have_codling_moth_trigger = yes
	}

	incompatible_with = frost
	incompatible_with = wildfire
	incompatible_with = extreme_winds
	incompatible_with = hailstorm
	incompatible_with = torrential_rains
	# Incompatible with extreme weather

	range = {
		integer_range = {
			min = 3
			max = 3
		}
	}

	duration = {
		fixed_range = {
			min = 250
			max = 500
		}
	}

	modifier = {
		#Basic Codling Moth Effects
		goods_output_fruit_mult = -0.25
	}

	intensity = {
		fixed_range = {
			min = 1
			max = 1
		}
	}

	chance = {
		value = 0.0025
		if = {
			limit = {
				any_scope_state = {
					any_scope_building = {
						is_building_type = building_fruit_orchard
						level >= 5
					}
				}
			}
			multiply = 2
		}
		if = {
			limit = {
				any_scope_state = {
					any_scope_building = {
						is_building_type = building_fruit_orchard
						level >= 10
					}
				}
			}
			multiply = 2
		}
		if = {
			limit = {
				any_scope_state = {
					any_scope_building = {
						is_building_type = building_fruit_orchard
						level >= 15
					}
				}
			}
			multiply = 2
		}
		if = {
			limit = {
				any_scope_state = {
					any_scope_building = {
						is_building_type = building_fruit_orchard
						level >= 20
					}
				}
			}
			multiply = 5
		}
	}

	color = { 122 255 89 }
	icon = "gfx/interface/icons/harvest_condition_icons/theiler_condition_codling_moth.dds"
	incompatible_terrain = { }
	map_texture = "gfx/map/map_painting/harvest_conditions_map_patterns/harvest_conditions_bad.dds"
	graphics = locust_swarm
}
