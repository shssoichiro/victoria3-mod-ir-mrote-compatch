# Madman at the Tank Factory in [SCOPE.sS('tank_state').GetName]
MFE_trade_good_events.6 = {
	dlc = dlc_morgenrote
	type = country_event
	placement = scope:tank_state
	title = "MFE_trade_good_events.6.t"
	desc = "MFE_trade_good_events.6.d"
	flavor = "MFE_trade_good_events.6.f"
	duration = 3
	icon = "gfx/interface/icons/event_icons/mobile_armor.dds"
	minor_left_icon = g:tanks
	minor_right_icon = g:tanks
	event_image = {
		video = "unspecific_devastation"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/destruction"
	cooldown = {
		months = very_long_cooldown
	}
	trigger = {
		any_scope_building = {
			is_building_type = building_war_machine_industry
			occupancy >= 0.5
			level >= 1
			level < 7
		}
	}
	immediate = {
		random_scope_building = {
			limit = {
				is_building_type = building_war_machine_industry
				occupancy >= 0.5
				level >= 1
				level < 7
			}
			save_scope_as = tank_factory
			state = {
				save_scope_as = tank_state
			}
		}
	}
	option = {
		name = MFE_trade_good_events.6.a
		default_option = yes
		# Hopefully they get over it
		scope:tank_state = {
			add_modifier = {
				name = MFE_tank_factory_destruction_mod
				days = normal_modifier_time
			}
		}
		scope:tank_factory = {
			add_modifier = {
				name = MFE_tank_factory_destruction_building_mod
				days = normal_modifier_time
			}
		}
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = MFE_trade_good_events.6.b
		scope:tank_state = {
			add_modifier = {
				name = MFE_tank_factory_destruction_aid_sent_mod
				days = short_modifier_time
			}
		}
		scope:tank_factory = {
			add_modifier = {
				name = MFE_tank_factory_destruction_building_mod
				days = short_modifier_time
			}
		}
		add_modifier = {
			name = MFE_sent_tank_destruction_aid_mod
			days = short_modifier_time
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					net_fixed_income > 2000
				}
				add = 2
			}
		}
	}
}

# Hardwood in [SCOPE.sS('hardwood_state').GetName] is extremely hard
MFE_trade_good_events.14 = {
	dlc = dlc_morgenrote

	type = country_event
	placement = scope:hardwood_state
	title = "MFE_trade_good_events.14.t"
	desc = "MFE_trade_good_events.14.d"
	flavor = "MFE_trade_good_events.14.f"
	duration = 3
	icon = "gfx/interface/icons/event_icons/unit_attack.dds"
	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/gold_prospectors"

	cooldown = { months = very_long_cooldown }
	trigger = {
		any_scope_building = {
			is_building_type = building_logging_camp
			modifier:goods_output_hardwood_add > 0
			level >= 5
		}
	}

	immediate = {
		random_scope_building = {
			limit = {
				is_building_type = building_logging_camp
				modifier:goods_output_hardwood_add > 0
				level >= 5
			}
			state = {
				save_scope_as = hardwood_state
			}
		}
	}

	option = {
		name = MFE_trade_good_events.14.a
		default_option = yes
		scope:hardwood_state = {
			random_scope_building = {
				limit = {
					is_building_type = building_logging_camp
					modifier:goods_output_hardwood_add > 0
					level >= 5
				}

				add_modifier = {
					name = MFE_very_hard_wood_mod
					days = normal_modifier_time
				}
			}
		}
	}

	option = {
		name = MFE_trade_good_events.14.b
		default_option = yes
		trigger = {
			any_scope_building = {
				is_building_type = building_instrument_workshops
			}
		}
		random_scope_building = {
			limit = {
				is_building_type = building_instrument_workshops
			}
			add_modifier = {
				name = MFE_great_instruments_mod
				days = normal_modifier_time
			}
		}
	}
}
